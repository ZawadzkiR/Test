import io
import os
import re
import sys
import html
import time
import queue
import getpass
import logging
import threading
import builtins
from contextlib import redirect_stdout, redirect_stderr

import snowflake.connector

try:
    from IPython.display import display, HTML, clear_output
except Exception:
    display = None
    HTML = None
    clear_output = None


def _is_notebook() -> bool:
    try:
        from IPython import get_ipython
        shell = get_ipython()
        return shell is not None and shell.__class__.__name__ == "ZMQInteractiveShell"
    except Exception:
        return False


def _clear_screen_secure() -> None:
    if _is_notebook() and clear_output:
        clear_output(wait=True)
        return

    if sys.stdout.isatty():
        # czyści ekran i zwykle także scrollback w wielu terminalach
        sys.stdout.write("\033[3J\033[2J\033[H")
        sys.stdout.flush()


def _show_login_link(url: str) -> None:
    if _is_notebook() and display and HTML:
        safe = html.escape(url, quote=True)
        display(HTML(f'<a href="{safe}" target="_blank" rel="noopener noreferrer">Otwórz link logowania Snowflake</a>'))
    else:
        print("\nLink logowania Snowflake:")
        print(url)
        print("")


class _RedactingCapture(io.TextIOBase):
    """
    Przechwytuje output connectora:
    - wyciąga pierwszy URL logowania SSO
    - nie wypisuje nic na stdout/stderr
    - ignoruje localhost redirect i prompty
    """
    URL_RE = re.compile(r"https?://[^\s'\"<>]+")
    LOCALHOST_RE = re.compile(r"^https?://(localhost|127\.0\.0\.1)(:\d+)?/?.*", re.IGNORECASE)

    def __init__(self):
        super().__init__()
        self.buffer = ""
        self.sso_url = None

    def writable(self):
        return True

    def write(self, s):
        if not isinstance(s, str):
            s = str(s)
        self.buffer += s

        for m in self.URL_RE.finditer(self.buffer):
            url = m.group(0)
            if self.LOCALHOST_RE.match(url):
                continue
            if self.sso_url is None:
                self.sso_url = url

        # nie wypisujemy nic dalej
        if len(self.buffer) > 16384:
            self.buffer = self.buffer[-4096:]
        return len(s)

    def flush(self):
        return None


def connect_snowflake_externalbrowser_safe(
    *,
    user: str,
    account: str,
    warehouse: str | None = None,
    database: str | None = None,
    schema: str | None = None,
    role: str | None = None,
    timeout_show_link: float = 15.0,
    **kwargs,
):
    """
    Bezpieczniejsze połączenie externalbrowser:
    - nie pokazuje surowych URL-i w logach stdout/stderr
    - pokazuje URL tylko kontrolowanie przez ten wrapper
    - końcowy localhost URL pobiera przez getpass()
    - po wpisaniu czyści widok
    """

    # 1) wyciszenie logów bibliotek
    for name in [
        "snowflake",
        "snowflake.connector",
        "botocore",
        "boto3",
        "urllib3",
        "requests",
    ]:
        logger = logging.getLogger(name)
        logger.handlers = []
        logger.propagate = False
        logger.setLevel(logging.CRITICAL)

    capture = _RedactingCapture()
    original_input = builtins.input
    result_q = queue.Queue()

    def _secure_input(prompt: str = ""):
        prompt_l = (prompt or "").lower()
        if "enter the url the sso url redirected you to" in prompt_l:
            redirect_url = getpass.getpass("Wklej końcowy redirect URL (localhost:...): ").strip()
            _clear_screen_secure()
            return redirect_url
        return original_input(prompt)

    def _worker():
        try:
            with redirect_stdout(capture), redirect_stderr(capture):
                conn = snowflake.connector.connect(
                    user=user,
                    account=account,
                    authenticator="externalbrowser",
                    warehouse=warehouse,
                    database=database,
                    schema=schema,
                    role=role,
                    **kwargs,
                )
            result_q.put(("ok", conn))
        except Exception as e:
            result_q.put(("err", e))

    try:
        builtins.input = _secure_input

        t = threading.Thread(target=_worker, daemon=True)
        t.start()

        # 2) czekamy chwilę, aż connector wygeneruje URL
        started = time.time()
        shown = False
        while t.is_alive():
            if capture.sso_url and not shown:
                _show_login_link(capture.sso_url)
                shown = True
            if not shown and (time.time() - started) > timeout_show_link:
                print("Trwa inicjalizacja logowania Snowflake...")
                shown = True
            try:
                status, payload = result_q.get_nowait()
                if status == "ok":
                    _clear_screen_secure()
                    print("Połączenie ze Snowflake zostało zestawione.")
                    return payload
                raise payload
            except queue.Empty:
                time.sleep(0.1)

        # odbiór wyniku po zakończeniu wątku
        status, payload = result_q.get(timeout=1)
        if status == "ok":
            _clear_screen_secure()
            print("Połączenie ze Snowflake zostało zestawione.")
            return payload
        raise payload

    finally:
        builtins.input = original_input
        # aktywne wymazanie przechwyconego bufora
        capture.buffer = ""
        capture.sso_url = None
