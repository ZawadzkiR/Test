import os
import sys
import builtins
import getpass
import logging

import snowflake.connector


def _detect_runtime():
    """
    Zwraca:
    - 'notebook' dla Jupyter/IPython notebook
    - 'terminal' dla zwykłego terminala
    """
    try:
        from IPython import get_ipython
        shell = get_ipython()
        if shell is not None:
            shell_name = shell.__class__.__name__
            if shell_name == "ZMQInteractiveShell":
                return "notebook"
    except Exception:
        pass

    return "terminal"


def _clear_runtime_view(runtime: str):
    if runtime == "notebook":
        try:
            from IPython.display import clear_output
            clear_output(wait=True)
            return
        except Exception:
            pass

    if os.name == "nt":
        os.system("cls")
    else:
        # reset zwykle czyści ekran lepiej niż clear
        os.system("reset")


def connect_snowflake_externalbrowser_safe(
    *,
    user: str,
    account: str,
    warehouse: str = None,
    database: str = None,
    schema: str = None,
    role: str = None,
    clear_after_auth: bool = True,
    **kwargs
):
    """
    Jedna funkcja dla terminala i Jupytera.
    
    Działanie:
    - externalbrowser pokazuje pierwszy link logowania normalnie
    - końcowy localhost URL jest pobierany przez getpass()
    - po wpisaniu czyści widok
    """

    runtime = _detect_runtime()
    original_input = builtins.input

    # wyciszenie loggerów bibliotek
    for name in [
        "snowflake",
        "snowflake.connector",
        "botocore",
        "boto3",
        "urllib3",
        "requests",
    ]:
        logger = logging.getLogger(name)
        logger.handlers = []
        logger.propagate = False
        logger.setLevel(logging.CRITICAL)

    def _secure_input(prompt=""):
        prompt_l = (prompt or "").lower()

        if "enter the url the sso url redirected you to" in prompt_l:
            if runtime == "notebook":
                print("Wklej końcowy redirect URL z localhost.")
            else:
                print("\nWklej końcowy redirect URL z localhost.")

            value = getpass.getpass("> ").strip()

            if clear_after_auth:
                _clear_runtime_view(runtime)

            return value

        return original_input(prompt)

    try:
        builtins.input = _secure_input

        conn = snowflake.connector.connect(
            user=user,
            account=account,
            authenticator="externalbrowser",
            warehouse=warehouse,
            database=database,
            schema=schema,
            role=role,
            **kwargs,
        )

        print("Połączenie ze Snowflake zostało zestawione.")
        return conn

    finally:
        builtins.input = original_input
